""" # 1. Variables Base extraídas del formulario Pa_val = float(self.precio_actual) Vps = float(self.variacion_precio) if float(self.variacion_precio) != 0 else 1.0 Sp = float(self.sub_mensuales_perdidas_max) Sm = float(self.sub_mensuales_nuevas_max) Up = float(self.tasa_perdida_base) # 2. Cálculos Intermedios Directos tpn_val = Sp / Vps # Intentamos buscar el mes anterior en DB para calcular Vtf real, si no, usamos el estándar (0.016) mes_anterior = Mes.objects.filter(num_mes=self.num_mes - 1).first() vtf_val = (float(self.tasa_fuga) - float(mes_anterior.tasa_fuga)) if mes_anterior else 0.016 per_val = vtf_val / Vps # 3. Modelado Simbólico con SymPy (Para obtener la 1era Derivada del Ingreso) Pa = sp.Symbol('Pa') Adq = Sm - tpn_val * Pa Chr = Up + per_val * Pa U = Adq / Chr I = Pa * U # Función de Ingresos Totales: Pa * U(Pa) # Primera derivada de la función de Ingresos con respecto a Pa dI_dPa = sp.diff(I, Pa) # Resolver dI/dPa = 0 para encontrar el precio que maximiza las ganancias soluciones = sp.solve(dI_dPa, Pa) precios_viables = [float(sol) for sol in soluciones if sol.is_real and sol > 0] precio_optimo_final = precios_viables[0] if precios_viables else Pa_val # Evaluar las funciones con el precio actual ingresado por el usuario adq_actual = float(Adq.subs(Pa, Pa_val)) chr_actual = float(Chr.subs(Pa, Pa_val)) u_actual = float(U.subs(Pa, Pa_val)) ingresos_actual = float(I.subs(Pa, Pa_val)) # 4. Guardar o actualizar en la base de datos (PrediccionMes) prediccion, created = models.PrediccionMes.objects.update_or_create( mes=self, defaults={ 'tasa_perdida_nuevos_no_suscritos': tpn_val, 'tasa_perdida_antiguos_suscritos': per_val, 'f_adquisicion': adq_actual, 'f_churn': chr_actual, 'usuarios_estables': int(u_actual), 'ingresos_totales': ingresos_actual, 'precio_optimizado_derivada': precio_optimo_final } ) return prediccion def ver_mes(request, id): mes_consultado = models.Mes.objects.get(id=id) return render(request, 'Elementos/Graficos/detalle_mes.html', {'mesKey': mes_consultado}) # Una función de Python normal para calcular la Adquisición def calcular_adquisicion(sm, tpn, pa): return sm - (tpn * pa) # Tu vista de Django def reporte_prediccion(request): # 1.
